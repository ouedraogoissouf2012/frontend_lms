const handleJoinVisio = async (seance) => {
  try {
    console.log('[VISIO] Rejoindre visio:', seance.id)

    // Créer une instance du composable pour cette séance
    currentSeanceId.value = seance.id
    visioParticipation.value = useVisioParticipation(seance.id)

    // Générer le lien Jitsi
    const user = auth.getUser()
    const link = `https://meet.jit.si/${seance.visio_room_id}#config.prejoinConfig.enabled=false&userInfo.displayName=${encodeURIComponent(user.name)}`

    // Utiliser le composable pour rejoindre (enregistrement + heartbeat automatique)
    await visioParticipation.value.joinVisio(link)

    console.log('[VISIO] ✅ Participation enregistrée, heartbeat démarré')
    $toast?.success('Vous avez rejoint la visioconférence')
  } catch (error) {
    console.error('[ERREUR] Join visio:', error)
    $toast?.error('Impossible de rejoindre la visioconférence')
  }
}
